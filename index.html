⚙ Configuration — Google Sheets

Suis ces 3 étapes une seule fois. Ensuite, toutes les tablettes des cadres synchronisent automatiquement vers le même tableur.

1 Créer le Google Sheet

Va sur sheets.google.com, crée un nouveau tableur et nomme-le Qualifications Moncor 2026. Garde-le ouvert.

2 Coller le script

Dans le tableur : menu Extensions → Apps Script. Supprime tout le contenu existant, colle le code ci-dessous, puis clique Déployer → Nouveau déploiement (type : Application Web, Accès : Tout le monde).

function doPost(e) {
  try {
    var data = JSON.parse(e.postData.contents);
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var sheet = ss.getSheetByName("Qualifications") || ss.insertSheet("Qualifications");
    if (sheet.getLastRow() === 0) {
      sheet.appendRow(["Timestamp","N° personnel","Nom","Prénom","Ancienneté",
        "Grade","Commune","Présence","Période","Évaluateur",
        "Engagement","Opérationnel","Comportement","Progression","Note globale",
        "Points forts","Axes amélioration","Notes détail"]);
    }
    var existing = sheet.getDataRange().getValues();
    var rowIdx = -1;
    // Si forceNew = true, on ajoute toujours une nouvelle ligne (historique par exercice)
    if (!data.forceNew) {
      for (var i = 1; i < existing.length; i++) {
        if (existing[i][1] === data.id && existing[i][8] === data.periode) { rowIdx = i + 1; break; }
      }
    }
    var row = [
      new Date().toLocaleString("fr-CH"), data.id, data.nom, data.prenom, data.anc,
      data.grade||"", data.commune||"", data.presence||"", data.periode||"", data.evaluateur||"",
      data.eng||"", data.ops||"", data.beh||"", data.prg||"", data.global||"",
      data.forts||"", data.axes||"", JSON.stringify(data.notes||{})
    ];
    if (rowIdx > 0) {
      sheet.getRange(rowIdx, 1, 1, row.length).setValues([row]);
    } else {
      sheet.appendRow(row);
    }
    return ContentService.createTextOutput(JSON.stringify({ok:true})).setMimeType(ContentService.MimeType.JSON);
  } catch(err) {
    return ContentService.createTextOutput(JSON.stringify({ok:false,error:err.toString()})).setMimeType(ContentService.MimeType.JSON);
  }
}

function doGet(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName("Qualifications");
  if (!sheet || sheet.getLastRow() < 2) {
    return ContentService.createTextOutput(JSON.stringify([])).setMimeType(ContentService.MimeType.JSON);
  }
  var data = sheet.getDataRange().getValues();
  var headers = data[0];
  var result = [];
  for (var i = 1; i < data.length; i++) {
    var obj = {};
    for (var j = 0; j < headers.length; j++) obj[headers[j]] = data[i][j];
    result.push(obj);
  }
  return ContentService.createTextOutput(JSON.stringify(result)).setMimeType(ContentService.MimeType.JSON);
}
